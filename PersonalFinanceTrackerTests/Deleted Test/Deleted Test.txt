private static IEnumerable<TestCaseData> LoadTextRowData()
{

    string filePath = Path.Combine("E:\\Chirag\\Learning\\TrueCoders\\repos\\Project_PersonalFinanceTracker_Research", "ForTransactionControllerRowTestValues.txt");
    int lineNumber = 0;

    if (!File.Exists(filePath))
    {
        throw new FileNotFoundException($"Critical Error: Test data file missing at {filePath}");
    }

    foreach (var line in File.ReadLines(filePath))
    {
        lineNumber++;
        if (string.IsNullOrWhiteSpace(line) || line.StartsWith("#")) continue; // Skip empty lines and comments

        // Using Pipe '|' instead of comma ',' to protect the text description field
        var parts = line.Split(',');

        // Parse every value safely, mapping empty strings to null for nullable types
        bool isFirstRequest = true;
        int dateRanges = 0;
        int? listIndexForCategories = ParseNullableInt(parts[0]);
        int? listIndexForAccounts = ParseNullableInt(parts[1]);
        string? transactionFilterDescription = string.IsNullOrWhiteSpace(parts[2]) ? null : parts[2].Trim();
        int? listIndexForTags = ParseNullableInt(parts[3]);
        int? listIndexForGoals = ParseNullableInt(parts[4]);
        byte transactionFilterCategorization = byte.Parse(parts[5].Trim());
        byte transactionFilterType = byte.Parse(parts[6].Trim());

        // Build the test case and name it dynamically for the Test Explorer tree
        var testCase = new TestCaseData(
            isFirstRequest, dateRanges, listIndexForCategories, listIndexForAccounts,
            transactionFilterDescription, listIndexForTags, listIndexForGoals,
            transactionFilterCategorization, transactionFilterType
        ).SetName($"Row_{lineNumber}_{transactionFilterDescription ?? "NoDescription"}");

        yield return testCase;
    }
}

// Helper method to deal with empty columns mapping to null values
private static int? ParseNullableInt(string value)
{
    return value == "null" ? null : int.Parse(value.Trim());
}
